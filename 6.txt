---------------------------------------------------
-- Create Department table
---------------------------------------------------

CREATE TABLE Department (
dept_id INT PRIMARY KEY,
dept_name VARCHAR(50),
location VARCHAR(50)
);

---------------------------------------------------
-- Create Employee table
---------------------------------------------------

CREATE TABLE Employee (
emp_id INT PRIMARY KEY,
emp_name VARCHAR(50),
salary DECIMAL(10,2),
dept_id INT,
FOREIGN KEY (dept_id) REFERENCES Department(dept_id)
);

---------------------------------------------------
-- a) Find employees who work in IT department
---------------------------------------------------

SELECT emp_id, emp_name, salary, dept_id
FROM Employee
WHERE dept_id = (
SELECT dept_id
FROM Department
WHERE dept_name = 'IT'
);

---------------------------------------------------
-- b) Employees whose salary is greater than
-- average salary of all employees
---------------------------------------------------

SELECT *
FROM Employee
WHERE salary > (
SELECT AVG(salary)
FROM Employee
);

---------------------------------------------------
-- c) Employees who work in same department as John
---------------------------------------------------

SELECT *
FROM Employee
WHERE dept_id = (
SELECT dept_id
FROM Employee
WHERE emp_name = 'John'
);

---------------------------------------------------
-- d) Department name of employee having
-- highest salary
---------------------------------------------------

SELECT dept_name
FROM Department
WHERE dept_id = (
SELECT dept_id
FROM Employee
WHERE salary = (
SELECT MAX(salary)
FROM Employee
)
);

---------------------------------------------------
-- e) Employees working in departments
-- located in Mumbai using IN
---------------------------------------------------

SELECT *
FROM Employee
WHERE dept_id IN (
SELECT dept_id
FROM Department
WHERE location = 'Mumbai'
);

---------------------------------------------------
-- f) Employees earning more than ANY employee
-- in department 2
---------------------------------------------------

SELECT *
FROM Employee
WHERE salary > ANY (
SELECT salary
FROM Employee
WHERE dept_id = 2
);
